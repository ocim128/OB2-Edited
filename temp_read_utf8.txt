@echo off
setlocal EnableExtensions EnableDelayedExpansion
title OpenBullet2 Testing Build
color 0A

echo(
echo ================================================
echo  OpenBullet2.Native - Incremental Testing Build
echo ================================================
echo(

set "SCRIPT_DIR=%~dp0"
set "PROJECT_DIR=%SCRIPT_DIR%.."
set "BUILD_DIR=%SCRIPT_DIR%testing"
set "USERDATA_SOURCE=%PROJECT_DIR%\UserData"
set "USERDATA_BUILD=%BUILD_DIR%\UserData"
set "USERDATA_BACKUP=%SCRIPT_DIR%UserData_backup"
set "CONFIGURATION=Testing"
set "RUNTIME=win-x64"
set "CLEAN_BUILD=0"
set "RUN_RESTORE=0"
set "SKIP_LAUNCH=0"
set "TOTAL_STEPS=8"
set /a STEP=1
set "START_TIME=%time%"

:parseArgs
if "%~1"=="" goto argsDone
if /I "%~1"=="--clean" set "CLEAN_BUILD=1"
if /I "%~1"=="-c" set "CLEAN_BUILD=1"
if /I "%~1"=="--restore" set "RUN_RESTORE=1"
if /I "%~1"=="-r" set "RUN_RESTORE=1"
if /I "%~1"=="--no-launch" set "SKIP_LAUNCH=1"
shift
goto parseArgs
:argsDone

set "NO_RESTORE_FLAG="
if "%RUN_RESTORE%"=="1" set "NO_RESTORE_FLAG=--no-restore"

call :PrintStep "Terminating existing OpenBullet2.Native processes..."
taskkill /f /im "OpenBullet2.Native.exe" >nul 2>&1
if errorlevel 1 (
    echo      No running OpenBullet2.Native processes found.
) else (
    echo      Terminated running OpenBullet2.Native processes.
)

call :PrintStep "Backing up existing UserData from build directory..."
if exist "%USERDATA_BUILD%" (
    if exist "%USERDATA_BACKUP%" rmdir /s /q "%USERDATA_BACKUP%"
    robocopy "%USERDATA_BUILD%" "%USERDATA_BACKUP%" /E /NP /NDL /NJH /NJS >nul
    echo      UserData backed up to bin\UserData_backup.
) else (
    echo      No existing UserData in build directory.
)

call :PrintStep "Preparing build directory (incremental by default)..."
if not exist "%BUILD_DIR%" mkdir "%BUILD_DIR%"
if "%CLEAN_BUILD%"=="1" (
    echo      Clean flag detected: running dotnet clean.
    pushd "%PROJECT_DIR%"
    dotnet clean "OpenBullet2.Native.csproj" --configuration %CONFIGURATION% --verbosity quiet >nul
    if errorlevel 1 (
        popd
        echo      Clean failed. See output above.
        goto :fail
    )
    popd
) else (
    echo      Using incremental build artifacts. Pass --clean to force a rebuild.
)
if "%RUN_RESTORE%"=="1" (
    echo      Running dotnet restore before publish.
    pushd "%PROJECT_DIR%"
    dotnet restore "OpenBullet2.Native.csproj" --verbosity quiet --nologo >nul
    if errorlevel 1 (
        popd
        echo      Restore failed. See output above.
        goto :fail
    )
    popd
) else (
    echo      Skipping explicit restore. Publish will restore if required.
)

call :PrintStep "Publishing OpenBullet2.Native (%CONFIGURATION% / %RUNTIME%)..."
pushd "%PROJECT_DIR%"
dotnet publish "OpenBullet2.Native.csproj" --configuration %CONFIGURATION% --runtime %RUNTIME% --self-contained false --output "%BUILD_DIR%" --verbosity quiet --nologo %NO_RESTORE_FLAG% >nul 2>&1
if errorlevel 1 (
    echo      Publish failed with quiet output. Retrying with diagnostics...
    dotnet publish "OpenBullet2.Native.csproj" --configuration %CONFIGURATION% --runtime %RUNTIME% --self-contained false --output "%BUILD_DIR%" --verbosity minimal --nologo %NO_RESTORE_FLAG%
    if errorlevel 1 (
        popd
        goto :failBuild
    )
)
popd
echo      Build completed successfully.

call :PrintStep "Copying UserData to build directory..."
if exist "%USERDATA_SOURCE%" (
    robocopy "%USERDATA_SOURCE%" "%USERDATA_BUILD%" /E /NP /NDL /NJH /NJS >nul
    echo      UserData copied to build directory.
) else (
    echo      No UserData found in project directory.
)

call :PrintStep "Restoring backed up UserData if available..."
if exist "%USERDATA_BACKUP%" (
    robocopy "%USERDATA_BACKUP%" "%USERDATA_BUILD%" /E /NP /NDL /NJH /NJS >nul
    echo      Backed up UserData restored.
) else (
    echo      No backup UserData to restore.
)

call :PrintStep "Verifying output and pruning non-Windows runtimes..."
if not exist "%BUILD_DIR%\OpenBullet2.Native.exe" (
    echo      OpenBullet2.Native.exe not found in build output!
    echo      Build directory: %BUILD_DIR%
    dir /b "%BUILD_DIR%\*.exe"
    goto :fail
)
if exist "%BUILD_DIR%\runtimes" (
    for /d %%i in ("%BUILD_DIR%\runtimes\android-*" "%BUILD_DIR%\runtimes\ios-*" "%BUILD_DIR%\runtimes\iossimulator-*" "%BUILD_DIR%\runtimes\linux-*" "%BUILD_DIR%\runtimes\osx-*" "%BUILD_DIR%\runtimes\maccatalyst-*" "%BUILD_DIR%\runtimes\tvos-*" "%BUILD_DIR%\runtimes\tvossimulator-*" "%BUILD_DIR%\runtimes\browser-*") do (
        if exist "%%i" rmdir /s /q "%%i" >nul 2>&1
    )
    echo      Non-Windows runtimes removed.
) else (
    echo      No runtimes folder found.
)

call :PrintStep "Launching OpenBullet2.Native..."
call :LaunchApplication

goto :success

:LaunchApplication
if "%SKIP_LAUNCH%"=="1" (
    echo      Launch skipped ^(per --no-launch^).
) else (
    pushd "%BUILD_DIR%"
    start "" "OpenBullet2.Native.exe"
    popd
    echo      Application started.
)
exit /b 0

:failBuild
echo      Publish failed again. Please check the project configuration.
goto :fail

:fail
echo(
echo Build failed. See messages above.
echo(
echo Press any key to exit... closing automatically in 30 seconds
choice /d y /t 30 >nul
goto :end

:success
set "END_TIME=%time%"
echo(
echo ================================================
echo  Build Process Completed
echo  Started:  %START_TIME%
echo  Finished: %END_TIME%
echo  Output:   %BUILD_DIR%
echo  Config:   %CONFIGURATION% / %RUNTIME%
if "%CLEAN_BUILD%"=="1" (echo  Clean:    Forced) else (echo  Clean:    Incremental)
if "%RUN_RESTORE%"=="1" (echo  Restore:  Explicit) else (echo  Restore:  On-demand)
if "%SKIP_LAUNCH%"=="1" (echo  Launch:   Skipped) else (echo  Launch:   Started)
echo ================================================
echo(
echo Press any key to exit... closing automatically in 30 seconds
choice /d y /t 30 >nul

:end
endlocal
exit /b 0

:PrintStep
echo(
echo [!STEP!/%TOTAL_STEPS%] %~1
set /a STEP+=1
exit /b 0
